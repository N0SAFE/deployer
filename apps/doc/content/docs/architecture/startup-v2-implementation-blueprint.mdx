## Phase 0: Trigger System

### Step 0.1 — Trigger&lt;T&gt; class

**File**: `apps/api/src/core/modules/triggers/trigger.ts`

```typescript
import { Subject, firstValueFrom, type Observable } from 'rxjs';

export class Trigger<T extends object = object> {
  private readonly subject = new Subject<T>();
  private _resolved = false;
  private _value: T | null = null;

  get resolved(): boolean {
    return this._resolved;
  }

  get value(): T {
    if (!this._resolved || !this._value) {
      throw new Error('Trigger not yet resolved');
    }
    return this._value;
  }

  get asObservable(): Observable<T> {
    return this.subject.asObservable();
  }

  async waitFor(): Promise<T> {
    if (this._resolved && this._value) return this._value;
    return firstValueFrom(this.subject);
  }

  fire(value: T): void {
    if (this._resolved) return;
    this._value = value;
    this._resolved = true;
    this.subject.next(value);
    this.subject.complete();
  }
}
```

**Validation criteria**:
- [ ] `trigger.waitFor()` resolves when `trigger.fire(value)` is called
- [ ] `trigger.fire(value)` is idempotent (second call is no-op)
- [ ] `trigger.resolved` is `true` after `fire()`
- [ ] `trigger.value` returns the payload after `fire()`
- [ ] `trigger.asObservable` emits once then completes

## Phase 2: Sub-App Orchestrator

### Step 2.1 — SubAppOrchestrator

**File**: `apps/api/src/core/modules/sub-app-orchestrator/sub-app-orchestrator.service.ts`

```typescript
import { Injectable, Logger } from '@nestjs/common';
import { NestFactory } from '@nestjs/core';
import type { INestApplicationContext } from '@nestjs/common';
import { TriggerRegistry } from '@/core/modules/triggers/trigger-registry';
import type { SubApp } from '@/core/modules/sub-app-runner/sub-app.interface';
import { SubAppContextModule } from '@/core/modules/sub-app-runner/sub-app-context.module';

@Injectable()
export class SubAppOrchestrator {
  private readonly logger = new Logger(SubAppOrchestrator.name);
  private readonly activeSubApps = new Map<string, INestApplicationContext>();

  constructor(private readonly triggers: TriggerRegistry) {}

  async runAll(subApps: SubApp[]): Promise<void> {
    const ordered = this.topologicalSort(subApps);
    for (const subApp of ordered) {
      await this.runOne(subApp);
    }
    this.logger.log('✅ All sub-apps completed');
  }

  async runOne(subApp: SubApp): Promise<void> {
    this.logger.log(`▶ Starting sub-app: ${subApp.name}`);

    const app = await NestFactory.createApplicationContext(subApp.module, {
      abortOnError: false,
      snapshot: process.env.NODE_ENV !== 'production',
    });

    this.activeSubApps.set(subApp.name, app);

    const timeout = subApp.timeoutMs ?? 30_000;
    await this.waitForTrigger(subApp.trigger, timeout);

    this.logger.log(`✅ ${subApp.name} trigger fired`);

    await app.close();
    this.activeSubApps.delete(subApp.name);
    this.logger.log(`🧹 ${subApp.name} destroyed`);
  }

  private async waitForTrigger<T extends object>(
    trigger: Trigger<T>,
    timeoutMs: number,
  ): Promise<T> {
    return Promise.race([
      trigger.waitFor(),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error(`Trigger timed out after ${timeoutMs}ms`)), timeoutMs),
      ),
    ]);
  }

  private topologicalSort(subApps: SubApp[]): SubApp[] {
    const nameMap = new Map(subApps.map((s) => [s.name, s]));
    const inDegree = new Map<string, number>();
    const adjacency = new Map<string, string[]>();

    for (const app of subApps) {
      inDegree.set(app.name, 0);
      adjacency.set(app.name, []);
    }
    for (const app of subApps) {
      for (const dep of app.dependsOn ?? []) {
        adjacency.get(dep)?.push(app.name);
        inDegree.set(app.name, (inDegree.get(app.name) ?? 0) + 1);
      }
    }
    const queue: string[] = [];
    for (const [name, degree] of inDegree) {
      if (degree === 0) queue.push(name);
    }
    const sorted: SubApp[] = [];
    while (queue.length > 0) {
      const name = queue.shift()!;
      const app = nameMap.get(name);
      if (app) sorted.push(app);
      for (const neighbor of adjacency.get(name) ?? []) {
        const newDegree = (inDegree.get(neighbor) ?? 1) - 1;
        inDegree.set(neighbor, newDegree);
        if (newDegree === 0) queue.push(neighbor);
      }
    }
    return sorted;
  }
}
```

**Validation criteria**:
- [ ] `runOne()` creates NestJS app context, waits for trigger, destroys app
- [ ] `runAll()` runs sub-apps in topological order
- [ ] Timeout throws if trigger never fires
- [ ] `app.close()` is called even on timeout

## Phase 4: Rewrite AppModule

### Step 4.1 — AppModule with Trigger-based forRootAsync

```typescript
@Module({
  imports: [
    TriggersModule.forRoot(),

    ConfigModule.forRootAsync({
      inject: [TriggerRegistry],
      useFactory: async (t: TriggerRegistry) => {
        const p = await t.config.waitFor();
        return p.config;
      },
    }),

    DatabaseModule.forRootAsync({
      inject: [TriggerRegistry],
      useFactory: async (t: TriggerRegistry) => {
        const p = await t.database.waitFor();
        return { pool: p.pool, drizzle: p.drizzle };
      },
    }),

    AuthModule.forRootAsync({
      inject: [TriggerRegistry],
      useFactory: async (t: TriggerRegistry) => {
        const p = await t.auth.waitFor();
        return { instance: p.authInstance, coreService: p.authCoreService };
      },
    }),

    // Feature modules
    DockerModule.forRootAsync({
      inject: [TriggerRegistry],
      useFactory: async (t: TriggerRegistry) => {
        const [db, docker] = await Promise.all([
          t.database.waitFor(), t.docker.waitFor(),
        ]);
        return { pool: db.pool, drizzle: db.drizzle, dockerode: docker.dockerode };
      },
    }),

    // ORPC
    ORPCModule.forRootAsync({
      inject: [TriggerRegistry],
      useFactory: async (t: TriggerRegistry) => {
        const [db, auth] = await Promise.all([
          t.database.waitFor(), t.auth.waitFor(),
        ]);
        return {
          context: { auth: auth.coreService.createEmptyAuthUtils() },
        };
      },
    }),
  ],
  providers: [
    SubAppOrchestrator,
    {
      provide: 'APP_BOOTSTRAP',
      useFactory: async (orchestrator: SubAppOrchestrator) => {
        await orchestrator.runAll(ALL_SUB_APPS);
        return true;
      },
      inject: [SubAppOrchestrator],
    },
  ],
})
export class AppModule {}
```

### Test Fixture: Trigger

```typescript
// trigger.spec.ts
import { Trigger } from './trigger';

describe('Trigger', () => {
  it('resolves waitFor() when fire() is called', async () => {
    const trigger = new Trigger<{ value: string }>();
    const promise = trigger.waitFor();
    trigger.fire({ value: 'hello' });
    await expect(promise).resolves.toEqual({ value: 'hello' });
  });

  it('is idempotent', () => {
    const trigger = new Trigger<{ value: string }>();
    trigger.fire({ value: 'first' });
    trigger.fire({ value: 'second' }); // no-op
    expect(trigger.value).toEqual({ value: 'first' });
  });

  it('returns immediately if already resolved', async () => {
    const trigger = new Trigger<{ value: string }>();
    trigger.fire({ value: 'hello' });
    await expect(trigger.waitFor()).resolves.toEqual({ value: 'hello' });
  });

  it('throws when accessing value before fire', () => {
    const trigger = new Trigger<{ value: string }>();
    expect(() => trigger.value).toThrow('Trigger not yet resolved');
  });
});
```

// sub-app-orchestrator.spec.ts
import { Test } from '@nestjs/testing';
import { Module, Injectable, OnModuleInit, Inject } from '@nestjs/common';
import { SubAppOrchestrator } from './sub-app-orchestrator.service';
import { TriggerRegistry } from '@/core/modules/triggers/trigger-registry';
import { SUB_APP_CONTEXT, SubAppContextModule } from '@/core/modules/sub-app-runner';
import type { SubApp, SubAppContext } from '@/core/modules/sub-app-runner/sub-app.interface';

describe('SubAppOrchestrator', () => {
  let orchestrator: SubAppOrchestrator;
  let triggers: TriggerRegistry;

  beforeEach(async () => {
    const module = await Test.createTestingModule({
      providers: [SubAppOrchestrator, TriggerRegistry],
    }).compile();

    orchestrator = module.get(SubAppOrchestrator);
    triggers = module.get(TriggerRegistry);
  });

  it('runs a single sub-app and waits for trigger', async () => {
    let destroyed = false;

    @Injectable()
    class TestService implements OnModuleInit {
      constructor(@Inject(SUB_APP_CONTEXT) private ctx: SubAppContext<{ ok: boolean }>) {}
      async onModuleInit() { this.ctx.trigger.fire({ ok: true }); }
    }

    @Module({
      imports: [SubAppContextModule.forRoot({ trigger: triggers.config })],
      providers: [TestService],
    })
    class TestModule {}

    const subApp: SubApp<{ ok: boolean }> = {
      name: 'test',
      module: TestModule,
      trigger: triggers.config,
    };

    await orchestrator.runOne(subApp);
    expect(triggers.config.resolved).toBe(true);
    expect(triggers.config.value).toEqual({ ok: true });
  });

  it('runs sub-apps in dependency order', async () => {

    // Sub-app A: no deps

    @Module({ imports: [SubAppContextModule.forRoot({ trigger: triggers.config })], providers: [ServiceA] })

      { name: 'b', module: ModuleB, dependsOn: ['a'], trigger: triggers.database },
      { name: 'a', module: ModuleA, trigger: triggers.config },
    ];

    await orchestrator.runAll(subApps);
    expect(order).toEqual(['a', 'b']);
  });
});
